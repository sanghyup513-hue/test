import sys
p = sys.argv[1]
s = open(p).read()

if '[PATCH4]' in s:
    print("patch4 이미 적용됨"); sys.exit(0)

old = """                cmd = ['trtllm-serve']
                import os as _os
                _ext = _os.environ.get('MLPERF_GPU_LIST')
                if _ext:
                    env['CUDA_VISIBLE_DEVICES'] = _ext
                else:
                    start_gpu = index * gpus_per_server
                    gpu_ids = list(range(start_gpu, start_gpu + gpus_per_server))
                    env['CUDA_VISIBLE_DEVICES'] = ','.join(map(str, gpu_ids))"""

new = """                cmd = ['trtllm-serve']
                # [PATCH4] index 는 srun 스텝마다 항상 0 이라, 원본 계산식으로는
                # 모든 서버가 물리 GPU 0 을 가리켜 2번째부터 메모리 부족으로 죽습니다.
                # run_scaleout.sh 가 랭크별로 다르게 넣어주는 NVIDIA_VISIBLE_DEVICES
                # 를 우선 사용합니다. 컨테이너가 이미 내 몫만 보고 있으면(격리 성공)
                # 원본 계산이 맞으므로 그대로 둡니다.
                _nvd = os.getenv('NVIDIA_VISIBLE_DEVICES', '') or ''
                _mine = [t for t in _nvd.replace(' ', '').split(',') if t.isdigit()]
                try:
                    _vis = len(subprocess.check_output(
                        ['nvidia-smi', '--query-gpu=index', '--format=csv,noheader'],
                        text=True).strip().splitlines())
                except Exception:
                    _vis = 0
                if _mine and _vis > gpus_per_server:
                    gpu_ids = _mine[:gpus_per_server]
                    logging.info(f"[PATCH4] container sees {_vis} GPUs; "
                                 f"using NVIDIA_VISIBLE_DEVICES -> {gpu_ids}")
                else:
                    start_gpu = index * gpus_per_server
                    gpu_ids = list(range(start_gpu, start_gpu + gpus_per_server))
                env['CUDA_VISIBLE_DEVICES'] = ','.join(map(str, gpu_ids))"""

if old not in s:
    print("패턴 불일치 — 실제 코드를 확인하세요"); sys.exit(1)
open(p,'w').write(s.replace(old,new))
print("patch4 적용 완료 (MLPERF_GPU_LIST 잔재 제거 포함)")
PYEOF
